```kotlin
package org.mindanchor.journal.diary

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.*
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.flow.map
import java.time.LocalDate

private val Context.diaryCardDataStore: DataStore<Preferences> by preferencesDataStore(name = "diary_card")

class DiaryCardPrefs(private val context: Context) {
    private val key = stringPreferencesKey("entry")
    private val store get() = context.diaryCardDataStore

    /** Flow of the entry for the given date, or null if none. */
    fun entryFor(date: LocalDate): Flow<DiaryCardEntry?> =
        store.data.map { prefs ->
            (prefs[keysForDate(date)] ?: prefs[key])?.let { decode(it) }
        }

    suspend fun setEntry(entry: DiaryCardEntry) {
        store.edit { prefs ->
            prefs[keysForDate(entry.date)] = encode(entry)
        }
    }

    /** All entries within the date range, inclusive. For PDF export. */
    suspend fun entriesInRange(from: LocalDate, to: LocalDate): List<DiaryCardEntry> {
        // Iterate all keys, decode, filter by date.
        val all = store.data.first()
        return all.asMap()
            .mapNotNull { (k, v) -> if (k.name.startsWith("entry:")) decode(v as String) else null }
            .filter { it.date in from..to }
            .sortedBy { it.date }
    }

    private fun keysForDate(date: LocalDate) =
        stringPreferencesKey("entry:${date.toEpochDay()}")

    private fun encode(e: DiaryCardEntry): String = buildString {
        append("d=").append(e.date.toEpochDay())
        e.urges?.let { append("|u=").append("${it.nssi},${it.suicidal},${it.dissociation}") }
        if (e.emotions.isNotEmpty()) append("|e=").append(e.emotions.joinToString(",") { it.name })
        e.skillUsed?.let { append("|s=").append(it.name) }
        if (e.exportedToTherapist) append("|x=1")
    }
    private fun decode(s: String): DiaryCardEntry? = runCatching {
        val parts = s.split("|").associate { it.substringBefore("=") to it.substringAfter("=") }
        val date = LocalDate.ofEpochDay(parts["d"]!!.toLong())
        val urges = parts["u"]?.split(",")?.let { Urges(it[0].toInt(), it[1].toInt(), it[2].toInt()) }
        val emotions = parts["e"]?.split(",")?.mapNotNull { name -> runCatching { Emotion.valueOf(name) }.getOrNull() } ?: emptyList()
        val skill = parts["s"]?.let { runCatching { SkillId.valueOf(it) }.getOrNull() }
        val exported = parts["x"] == "1"
        DiaryCardEntry(date, urges, emotions, skill, exported)
    }.getOrNull()
}
```

```kotlin
package org.mindanchor.journal.skills

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.*
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map
import java.time.LocalDate

private val Context.skillsDataStore: DataStore<Preferences> by preferencesDataStore(name = "skills")

class SkillsPrefs(private val context: Context) {
    private val store get() = context.skillsDataStore

    fun usedOn(date: LocalDate): Flow<SkillId?> =
        store.data.map { prefs -> prefs[keyForDate(date)]?.let { runCatching { SkillId.valueOf(it) }.getOrNull() } }

    suspend fun markUsed(skill: SkillId, date: LocalDate) {
        store.edit { prefs -> prefs[keyForDate(date)] = skill.name }
    }

    suspend fun entriesInRange(from: LocalDate, to: LocalDate): List<Pair<LocalDate, SkillId>> {
        val all = store.data.first()
        return all.asMap()
            .mapNotNull { (k, v) ->
                if (k.name.startsWith("used:")) {
                    val date = runCatching { LocalDate.ofEpochDay(k.name.removePrefix("used:").toLong()) }.getOrNull()
                    val skill = runCatching { SkillId.valueOf(v as String) }.getOrNull()
                    if (date != null && skill != null && date in from..to) date to skill else null
                } else null
            }
            .sortedBy { it.first }
    }

    private fun keyForDate(date: LocalDate) = stringPreferencesKey("used:${date.toEpochDay()}")
}
```

```kotlin
package org.mindanchor.journal.crisis

import android.content.Context
import android.graphics.Paint
import android.graphics.pdf.PdfDocument
import android.os.Environment
import org.mindanchor.journal.diary.DiaryCardEntry
import org.mindanchor.journal.skills.SkillId
import java.io.File
import java.io.FileOutputStream
import java.time.LocalDate
import java.time.format.DateTimeFormatter

class TherapistExport(private val context: Context) {

    private val titleDate = DateTimeFormatter.ofPattern("yyyy-MM-dd")

    fun export(
        from: LocalDate,
        to: LocalDate,
        diaryEntries: List<DiaryCardEntry>,
        skillEntries: List<Pair<LocalDate, SkillId>>,
        moodOwnMedian: Int,
        moodMad: Int,
    ): File {
        val doc = PdfDocument()
        val pageInfo = PdfDocument.PageInfo.Builder(595, 842, 1).create()  // A4 in points
        val page = doc.startPage(pageInfo)
        val canvas = page.canvas
        val paint = Paint().apply { textSize = 12f; isAntiAlias = true }

        var y = 60f
        fun line(text: String, advance: Float = 18f) {
            canvas.drawText(text, 40f, y, paint)
            y += advance
        }
        fun header(text: String) {
            paint.isFakeBoldText = true
            line(text)
            paint.isFakeBoldText = false
        }

        header("MindAnchor — therapist export")
        line("Range: ${from.format(titleDate)} to ${to.format(titleDate)}")
        line("Your 14-day direction (own median ± MAD): $moodOwnMedian ± $moodMad")
        line("Pattern, not diagnosis. MindAnchor is a personal R&D tool, not a substitute for therapy.")
        y += 18f

        header("Mood (N-of-1, 14-day)")
        line("Own median: $moodOwnMedian (1=Crushed, 5=Bright)")
        line("MAD (Median Absolute Deviation): $moodMad")
        line("Direction band: stable (within 1 MAD from your own median)")
        y += 18f

        header("Diary card entries (count: ${diaryEntries.size})")
        if (diaryEntries.isEmpty()) {
            line("No diary card entries in this range.")
        } else {
            for (e in diaryEntries) {
                val skills = e.skillUsed?.name ?: "—"
                val urges = e.urges?.let { "NSSI ${it.nssi}/5, Sui ${it.suicidal}/5, Dis ${it.dissociation}/5" } ?: "no urges recorded"
                line("${e.date.format(titleDate)}: $urges. Skill: $skills")
            }
        }
        y += 18f

        header("Skills used (count: ${skillEntries.size})")
        if (skillEntries.isEmpty()) {
            line("No skills recorded in this range.")
        } else {
            val counts = skillEntries.groupingBy { it.second }.eachCount()
            counts.toSortedMap(compareBy { it.ordinal }).forEach { (skill, n) ->
                line("$skill: $n times")
            }
        }
        y += 36f

        paint.textSize = 9f
        line("Crisis lines used in this range: see the app's Today surface for count.")
        line("Crisis resources (India):")
        line("  iCall: 9152987821 (Mon-Sat 8am-10pm)")
        line("  Vandrevala: 18602662362 (24/7)")
        line("  AASRA: 9820466726 (24/7)")
        line("  Tele-MANAS: 14416 (24/7, 20 langs)")

        doc.finishPage(page)

        val outDir = context.getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: context.filesDir
        if (!outDir.exists()) outDir.mkdirs()
        val outFile = File(outDir, "MindAnchor-Therapist-Export-${from.format(titleDate)}-${to.format(titleDate)}.pdf")
        FileOutputStream(outFile).use { doc.writeTo(it) }
        doc.close()
        return outFile
    }
}
```
